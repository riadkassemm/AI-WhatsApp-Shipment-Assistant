  stop_clean() {
    systemctl stop --no-block "$1.service"
    for task_wait in $(seq 1 120); do
      task_state=$(systemctl show "$1.service" -p ActiveState --value)
      if test "$task_state" = inactive; then
        test "$(systemctl show "$1.service" -p MainPID --value)" = 0
        test "$(systemctl show "$1.service" -p Result --value)" = success
        if test "$1" != ngrok; then
          test "$(systemctl show "$1.service" -p ExecMainStatus --value)" = 0
        fi
        return
      fi
      test "$task_state" != failed || return 1
      sleep 1
    done
    printf 'STOP_UNRESOLVED: %s; do not kill or activate a replacement\n' "$1" >&2
    return 1
  }
  guard_shutdown() {
    for task_unit in whatsapp-bridge shipment-bot; do
      install -d -m0755 "/run/systemd/system/$task_unit.service.d"

    done
